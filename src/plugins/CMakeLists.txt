# Every effect and device lives in a folder of its own: its C++, its dialog and its tests, so that
# adding one is adding a folder rather than editing lists all over the tree.
#
# A plugin's files are built into the libraries and the QML module they always belonged to, so the
# layer rules still hold inside a folder and nothing else in the build has to know where it lives.

#! Adds the plugin in the calling folder.
#!
#! DOMAIN_SOURCES, APPLICATION_SOURCES and VIEW_SOURCES go into the library of that layer. QML_FILES
#! go into the QML module at view/qml/Dialogs/, where they always were: the dialogs import
#! "../Components" relatively, and lupdate names a dialog's translation context after its file, so
#! both keep working unchanged. Every folder in unit_tests/ that has a CMakeLists.txt is a test.
function(_noteahead_add_plugin)
    cmake_parse_arguments(PLUGIN "" "" "DOMAIN_SOURCES;APPLICATION_SOURCES;VIEW_SOURCES;QML_FILES" ${ARGN})
    if(PLUGIN_UNPARSED_ARGUMENTS)
        message(FATAL_ERROR "Unknown plugin arguments in ${CMAKE_CURRENT_SOURCE_DIR}: ${PLUGIN_UNPARSED_ARGUMENTS}")
    endif()
    if(PLUGIN_DOMAIN_SOURCES)
        target_sources(${DomainLibName} PRIVATE ${PLUGIN_DOMAIN_SOURCES})
    endif()
    if(PLUGIN_APPLICATION_SOURCES)
        target_sources(${ApplicationLibName} PRIVATE ${PLUGIN_APPLICATION_SOURCES})
    endif()
    if(PLUGIN_VIEW_SOURCES)
        target_sources(${ViewLibName} PRIVATE ${PLUGIN_VIEW_SOURCES})
    endif()
    foreach(QML_FILE ${PLUGIN_QML_FILES})
        get_filename_component(QML_FILE_NAME ${QML_FILE} NAME)
        set_source_files_properties(${QML_FILE} PROPERTIES QT_RESOURCE_ALIAS view/qml/Dialogs/${QML_FILE_NAME})
    endforeach()
    if(PLUGIN_QML_FILES)
        qt_target_qml_sources(${BINARY_NAME} QML_FILES ${PLUGIN_QML_FILES})
    endif()
    if(BUILD_TESTS)
        file(GLOB TEST_LISTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/unit_tests/*/CMakeLists.txt)
        foreach(TEST_LIST ${TEST_LISTS})
            get_filename_component(TEST_DIR ${TEST_LIST} DIRECTORY)
            add_subdirectory(${TEST_DIR})
        endforeach()
    endif()
endfunction()

function(noteahead_add_effect)
    _noteahead_add_plugin(${ARGN})
endfunction()

function(noteahead_add_device)
    _noteahead_add_plugin(${ARGN})
endfunction()
