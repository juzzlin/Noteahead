# Every effect and device lives in a folder of its own: its C++, its dialog and its tests, so that
# adding one is adding a folder rather than editing lists all over the tree.
#
# A plugin's files are built into the libraries and the QML module they always belonged to, so the
# layer rules still hold inside a folder and nothing else in the build has to know where it lives.

#! Adds the plugin in the calling folder.
#!
#! DOMAIN_SOURCES, APPLICATION_SOURCES and VIEW_SOURCES go into the library of that layer. QML_FILES
#! go into the QML module at view/qml/Dialogs/, where they always were: the dialogs import
#! "../Components" relatively, and lupdate names a dialog's translation context after its file, so
#! both keep working unchanged. Every folder in unit_tests/ that has a CMakeLists.txt is a test.
#!
#! A plugin registers itself when it has the files for it: <folder>_registration.cpp in
#! DOMAIN_SOURCES defines register<Folder><Kind>(), which adds it to its factory, and
#! <folder>_view_registration.cpp in VIEW_SOURCES defines register<Folder><Kind>View(), which adds
#! its gallery entry and dialog to the PluginCatalog. The calls to both are generated below.
function(_noteahead_add_plugin KIND)
    cmake_parse_arguments(PLUGIN "" "" "DOMAIN_SOURCES;APPLICATION_SOURCES;VIEW_SOURCES;QML_FILES" ${ARGN})
    if(PLUGIN_UNPARSED_ARGUMENTS)
        message(FATAL_ERROR "Unknown plugin arguments in ${CMAKE_CURRENT_SOURCE_DIR}: ${PLUGIN_UNPARSED_ARGUMENTS}")
    endif()
    if(PLUGIN_DOMAIN_SOURCES)
        target_sources(${DomainLibName} PRIVATE ${PLUGIN_DOMAIN_SOURCES})
    endif()
    if(PLUGIN_APPLICATION_SOURCES)
        target_sources(${ApplicationLibName} PRIVATE ${PLUGIN_APPLICATION_SOURCES})
    endif()
    if(PLUGIN_VIEW_SOURCES)
        target_sources(${ViewLibName} PRIVATE ${PLUGIN_VIEW_SOURCES})
    endif()
    # Collected rather than added here: see NOTEAHEAD_PLUGIN_QML_FILES in src/CMakeLists.txt
    foreach(QML_FILE ${PLUGIN_QML_FILES})
        set_property(GLOBAL APPEND PROPERTY NOTEAHEAD_PLUGIN_QML_FILES ${CMAKE_CURRENT_SOURCE_DIR}/${QML_FILE})
    endforeach()
    get_filename_component(FOLDER ${CMAKE_CURRENT_SOURCE_DIR} NAME)
    _noteahead_pascal_case(${FOLDER} PASCAL)
    if("${FOLDER}_registration.cpp" IN_LIST PLUGIN_DOMAIN_SOURCES)
        set_property(GLOBAL APPEND PROPERTY NOTEAHEAD_${KIND}_REGISTRATIONS register${PASCAL}${KIND})
    endif()
    if("${FOLDER}_view_registration.cpp" IN_LIST PLUGIN_VIEW_SOURCES)
        set_property(GLOBAL APPEND PROPERTY NOTEAHEAD_${KIND}_VIEW_REGISTRATIONS register${PASCAL}${KIND}View)
    endif()
    if(BUILD_TESTS)
        file(GLOB TEST_LISTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/unit_tests/*/CMakeLists.txt)
        foreach(TEST_LIST ${TEST_LISTS})
            get_filename_component(TEST_DIR ${TEST_LIST} DIRECTORY)
            add_subdirectory(${TEST_DIR})
        endforeach()
    endif()
endfunction()

function(noteahead_add_effect)
    _noteahead_add_plugin(Effect ${ARGN})
endfunction()

function(noteahead_add_device)
    _noteahead_add_plugin(Device ${ARGN})
endfunction()

#! eq_8_band_parametric -> Eq8BandParametric
function(_noteahead_pascal_case INPUT OUTPUT)
    string(REPLACE "_" ";" PARTS ${INPUT})
    set(RESULT "")
    foreach(PART ${PARTS})
        string(SUBSTRING ${PART} 0 1 FIRST)
        string(SUBSTRING ${PART} 1 -1 REST)
        string(TOUPPER ${FIRST} FIRST)
        string(APPEND RESULT ${FIRST}${REST})
    endforeach()
    set(${OUTPUT} ${RESULT} PARENT_SCOPE)
endfunction()

#! Writes a generated source only when its content changes, so that reconfiguring does not
#! rebuild what includes it.
function(_noteahead_write_generated FILE CONTENT)
    file(WRITE ${FILE}.tmp "${CONTENT}")
    configure_file(${FILE}.tmp ${FILE} COPYONLY)
endfunction()

#! The calls to every plugin's registration functions: one file per kind and layer, so that a
#! binary that only uses effects does not link every device through the same object file.
function(_noteahead_generate_registrations)
    set(HEADER "// Generated by src/plugins/CMakeLists.txt from the plugin folders. Do not edit.\n\n")
    foreach(KIND Effect Device)
        string(TOLOWER ${KIND} KIND_LOWER)
        get_property(DOMAIN_CALLS GLOBAL PROPERTY NOTEAHEAD_${KIND}_REGISTRATIONS)
        get_property(VIEW_CALLS GLOBAL PROPERTY NOTEAHEAD_${KIND}_VIEW_REGISTRATIONS)
        set(DOMAIN "${HEADER}#include \"domain/${KIND_LOWER}s/${KIND_LOWER}_factory.hpp\"\n\nnamespace noteahead {\n\n")
        foreach(CALL ${DOMAIN_CALLS})
            string(APPEND DOMAIN "void ${CALL}();\n")
        endforeach()
        string(APPEND DOMAIN "\nvoid ${KIND}Factory::registerPlugins()\n{\n")
        foreach(CALL ${DOMAIN_CALLS})
            string(APPEND DOMAIN "    ${CALL}();\n")
        endforeach()
        string(APPEND DOMAIN "}\n\n} // namespace noteahead\n")
        set(VIEW "${HEADER}#include \"view/plugins/plugin_catalog.hpp\"\n\nnamespace noteahead {\n\n")
        foreach(CALL ${VIEW_CALLS})
            string(APPEND VIEW "void ${CALL}();\n")
        endforeach()
        string(APPEND VIEW "\nvoid PluginCatalog::register${KIND}Plugins()\n{\n")
        foreach(CALL ${VIEW_CALLS})
            string(APPEND VIEW "    ${CALL}();\n")
        endforeach()
        string(APPEND VIEW "}\n\n} // namespace noteahead\n")
        _noteahead_write_generated(${CMAKE_CURRENT_BINARY_DIR}/${KIND_LOWER}_plugin_registrations.cpp "${DOMAIN}")
        _noteahead_write_generated(${CMAKE_CURRENT_BINARY_DIR}/${KIND_LOWER}_plugin_view_registrations.cpp "${VIEW}")
        target_sources(${DomainLibName} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/${KIND_LOWER}_plugin_registrations.cpp)
        target_sources(${ViewLibName} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/${KIND_LOWER}_plugin_view_registrations.cpp)
    endforeach()
endfunction()
